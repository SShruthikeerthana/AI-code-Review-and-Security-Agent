def perform_static_security_scan(source_code: str, language: str) -> List[dict]:
    """
    Dynamically analyzes any provided source code line-by-line using rule-based patterns
    to detect real OWASP Top 10 vulnerabilities with exact 1-based line numbers.
    """
    findings = []
    seen_lines_and_categories = set()

    lines = source_code.split("\n")
    for line_idx, raw_line in enumerate(lines, start=1):
        line = raw_line.strip()
        # Skip pure comment lines
        if line.startswith("#") or line.startswith("//") or line.startswith("/*") or line.startswith("*"):
            continue

        for rule in SECURITY_RULES:
            key = (line_idx, rule["category"])
        if key in seen_lines_and_categories:
                continue

            for pattern in rule["patterns"]:
        if re.search(pattern, raw_line):
                    findings.append({
                        "title": rule["title"],
                        "description": rule["description"],
                        "line_number": line_idx,
                        "severity": rule["severity"],
                        "category": rule["category"],
                        "cwe_id": rule["cwe_id"],
                        "owasp_category": rule["owasp_category"]
                    })
                    seen_lines_and_categories.add(key)
                    break

    return findings

def get_mock_security_vulnerabilities(source_code: str, language: str) -> List[dict]:
    """
    Dynamic fallback for security vulnerability scanning when LLM API key is not configured.
    """
    return perform_static_security_scan(source_code, language)

def scan_security_vulnerabilities(source_code: str, language: str) -> List[dict]:
    api_key = settings.gemini_api_key or os.environ.get("GEMINI_API_KEY")
    if not api_key:
        print("INFO: GEMINI_API_KEY not set. Using dynamic static security vulnerability analysis engine.")
        return perform_static_security_scan(source_code, language)

    try:
        genai.configure(api_key=api_key)
        
        model = genai.GenerativeModel(
            model_name=settings.llm_model,
            system_instruction=SYSTEM_PROMPT
        )
        
        prompt = f"Analyze the following {language} code for security vulnerabilities:\n\n{source_code}"
        
        response = model.generate_content(
            prompt,
            generation_config=genai.GenerationConfig(
                response_mime_type="application/json",
                response_schema=SecurityVulnerabilityResult
            )
        )
        
        # Parse output
        result = SecurityVulnerabilityResult.model_validate_json(response.text)
        return [finding.model_dump() for finding in result.findings]
    except Exception as e:
        print(f"WARNING: LLM security scan failed or unconfigured ({e}). Falling back to dynamic static security scanner.")
        return perform_static_security_scan(source_code, language)

    public static int binarySearch(int[] arr, int target) {
        int left = 0;
        int right = arr.length - 1;

        while (left <= right) {
            int mid = left + (right - left) / 2;

            if (arr[mid] == target) {
                return mid;
            } else if arr[mid] < target) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }

        return -1;
