    to detect real OWASP Top 10 vulnerabilities with exact 1-based line numbers.

        line = raw_line.strip()

            if key in seen_lines_and_categories:

                    findings.append({
                        "title": rule["title"],
                        "description": rule["description"],
                        "line_number": line_idx,
                        "severity": rule["severity"],
                        "category": rule["category"],
                        "cwe_id": rule["cwe_id"],
                        "owasp_category": rule["owasp_category"]
                    })
                    seen_lines_and_categories.add(key)
                    break

    return findings
